cmake_minimum_required(VERSION 3.20)
project(microduck_voice LANGUAGES CXX)

find_package(CURL 8.14 REQUIRED)
find_package(nlohmann_json 3.11 QUIET)
if(NOT TARGET nlohmann_json::nlohmann_json)
    find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
endif()

add_executable(microduck-voice src/main.cpp src/realtime.cpp src/audio.cpp)
target_compile_features(microduck-voice PRIVATE cxx_std_20)
target_link_libraries(microduck-voice PRIVATE CURL::libcurl)
if(TARGET nlohmann_json::nlohmann_json)
    target_link_libraries(microduck-voice PRIVATE nlohmann_json::nlohmann_json)
else()
    target_include_directories(microduck-voice SYSTEM PRIVATE "${NLOHMANN_JSON_INCLUDE_DIR}")
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(ALSA REQUIRED)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(SPEEXDSP QUIET IMPORTED_TARGET speexdsp)
    endif()
    target_sources(microduck-voice PRIVATE src/alsa_audio.cpp)
    target_link_libraries(microduck-voice PRIVATE ALSA::ALSA)
    target_compile_definitions(microduck-voice PRIVATE MICRODUCK_ALSA)
    if(TARGET PkgConfig::SPEEXDSP)
        target_link_libraries(microduck-voice PRIVATE PkgConfig::SPEEXDSP)
        target_compile_definitions(microduck-voice PRIVATE MICRODUCK_SPEEXDSP)
        target_sources(microduck-voice PRIVATE src/board_control.cpp)
        target_compile_definitions(microduck-voice PRIVATE MICRODUCK_VOICE_CONTROL)
        set_target_properties(microduck-voice PROPERTIES INSTALL_RPATH "\$ORIGIN/lib")
    endif()
endif()

option(MICRODUCK_TESTS "Build the duplex protocol and speech gate tests" OFF)
if(MICRODUCK_TESTS)
    enable_testing()
    add_executable(microduck-speech-gate-test tests/speech_gate.cpp)
    target_include_directories(microduck-speech-gate-test PRIVATE src)
    target_compile_features(microduck-speech-gate-test PRIVATE cxx_std_20)
    add_test(NAME microduck-speech-gate COMMAND microduck-speech-gate-test)

    add_executable(microduck-duplex-test tests/duplex_client.cpp src/realtime.cpp)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_sources(microduck-duplex-test PRIVATE src/board_control.cpp)
        target_compile_definitions(microduck-duplex-test PRIVATE MICRODUCK_MOCK_BOARD_CONTROL)
    endif()
    target_include_directories(microduck-duplex-test PRIVATE src)
    target_compile_features(microduck-duplex-test PRIVATE cxx_std_20)
    target_link_libraries(microduck-duplex-test PRIVATE CURL::libcurl)
    if(TARGET nlohmann_json::nlohmann_json)
        target_link_libraries(microduck-duplex-test PRIVATE nlohmann_json::nlohmann_json)
    else()
        target_include_directories(microduck-duplex-test SYSTEM PRIVATE "${NLOHMANN_JSON_INCLUDE_DIR}")
    endif()
    if(MSVC)
        target_compile_options(microduck-duplex-test PRIVATE /W4 /utf-8)
    else()
        target_compile_options(microduck-duplex-test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
if(MSVC)
    target_compile_options(microduck-voice PRIVATE /W4 /utf-8)
else()
    target_compile_options(microduck-voice PRIVATE -Wall -Wextra -Wpedantic)
endif()
